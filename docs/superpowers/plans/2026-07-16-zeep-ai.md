# zeep-ai Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build zeep-ai crate + @zeep/ai package for local AI CLI integration (detect CLIs, stream responses, cancel cleanly).

**Architecture:** New Rust crate `zeep-ai` with CLI detector, ProviderAdapter trait (5 CLIs), AIRunner with process-group isolation (`libc::setsid`/`killpg`). New npm package `@zeep/ai` with AIText utilities, 4 React components, useAI hook. Registered as Tauri plugin `"zeep-ai"`.

**Tech Stack:** Rust (tokio process, libc), React 18, TypeScript, Vitest

**Design Spec:** `docs/superpowers/specs/2026-07-16-zeep-ai-design.md`

## Global Constraints

- Prompt passed via stdin, never as shell argument (SEC-01)
- No API keys handled by zeep-ai (SEC-02)
- Output treated as untrusted — sanitize via MarkdownRenderer (SEC-03)
- Process group kill: SIGTERM → 2s timeout → SIGKILL
- Follow existing crate patterns: Builder API, TauriPlugin, permissions build.rs
- Follow existing package patterns: vitest.config.ts, test-setup.ts, barrel index.ts

```rust
use serde::Serialize;

const CANDIDATES: &[&str] = &["claude", "codex", "ollama", "llm", "opencode"];

#[derive(Debug, Clone, Serialize)]
pub struct DetectedProvider {
    pub name: String,
    pub binary: String,
    pub version: Option<String>,
}

pub fn detect() -> Vec<DetectedProvider> {
    CANDIDATES.iter().filter_map(|name| {
        let binary = which(name)?;
        let version = get_version(&binary);
        Some(DetectedProvider { name: name.to_string(), binary, version })
    }).collect()
}

fn which(name: &str) -> Option<String> {
    let output = std::process::Command::new("which").arg(name).output().ok()?;
    if !output.status.success() { return None; }
    Some(String::from_utf8(output.stdout).ok()?.trim().to_string())
}

fn get_version(binary: &str) -> Option<String> {
    let output = std::process::Command::new(binary).arg("--version").output().ok()?;
    let out = String::from_utf8(output.stdout).ok()?;
    out.lines().next().map(|s| s.to_string()).filter(|s| !s.is_empty())
}
```

use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct RunOpts {
    pub model: Option<String>,
    pub max_tokens: Option<u32>,
    pub system: Option<String>,
}

impl Default for RunOpts {
    fn default() -> Self {
        Self { model: None, max_tokens: None, system: None }
    }
}

pub trait ProviderAdapter: Send + Sync {
    fn name(&self) -> &'static str;
    fn build_args(&self, opts: &RunOpts) -> Vec<String>;
}

pub struct ClaudeAdapter;
impl ProviderAdapter for ClaudeAdapter {
    fn name(&self) -> &'static str { "claude" }
    fn build_args(&self, opts: &RunOpts) -> Vec<String> {
        let mut args = vec!["-p".into()];
        if let Some(system) = &opts.system {
            args.extend(["--system".into(), system.clone()]);
        }
        args
    }
}

pub struct CodexAdapter;
impl ProviderAdapter for CodexAdapter {
    fn name(&self) -> &'static str { "codex" }
    fn build_args(&self, _opts: &RunOpts) -> Vec<String> {
        vec!["-i".into()]
    }
}

pub struct OllamaAdapter;
impl ProviderAdapter for OllamaAdapter {
    fn name(&self) -> &'static str { "ollama" }
    fn build_args(&self, opts: &RunOpts) -> Vec<String> {
        vec!["run".into(), opts.model.as_deref().unwrap_or("llama3.2").into()]
    }
}

pub struct LlmAdapter;
impl ProviderAdapter for LlmAdapter {
    fn name(&self) -> &'static str { "llm" }
    fn build_args(&self, _opts: &RunOpts) -> Vec<String> {
        vec!["-t".into()]
    }
}

pub struct OpencodeAdapter;
impl ProviderAdapter for OpencodeAdapter {
    fn name(&self) -> &'static str { "opencode" }
    fn build_args(&self, _opts: &RunOpts) -> Vec<String> {
        vec!["-p".into()]
    }
}

pub fn get_adapter(name: &str) -> Option<Box<dyn ProviderAdapter>> {
    match name {
        "claude" => Some(Box::new(ClaudeAdapter)),
        "codex" => Some(Box::new(CodexAdapter)),
        "ollama" => Some(Box::new(OllamaAdapter)),
        "llm" => Some(Box::new(LlmAdapter)),
        "opencode" => Some(Box::new(OpencodeAdapter)),
        _ => None,
    }
}

pub fn all_adapters() -> Vec<Box<dyn ProviderAdapter>> {
    vec![
        Box::new(ClaudeAdapter),
        Box::new(CodexAdapter),
        Box::new(OllamaAdapter),
        Box::new(LlmAdapter),
        Box::new(OpencodeAdapter),
    ]
}

- [ ] **4.1 Write AIRunner struct**

```rust
use std::collections::HashMap;
use std::sync::atomic::{AtomicU64, Ordering};
use std::sync::{Arc, Mutex};
use tokio::io::AsyncBufReadExt;
use tokio::process::Command;
use serde::Serialize;

type RunId = u64;

struct RunState {
    child: tokio::process::Child,
    pgid: i32,
}

#[derive(Debug, Clone, Serialize)]
pub struct AiChunk {
    pub run_id: RunId,
    pub text: String,
}

#[derive(Debug, Clone, Serialize)]
pub struct AiDone {
    pub run_id: RunId,
}

pub struct AIRunner {
    next_id: AtomicU64,
    runs: Arc<Mutex<HashMap<RunId, RunState>>>,
}

impl AIRunner {
    pub fn new() -> Self {
        Self { next_id: AtomicU64::new(1), runs: Arc::new(Mutex::new(HashMap::new())) }
    }

    pub async fn run_stream(
        &self,
        binary: &str,
        args: &[String],
        prompt: &str,
        app_handle: tauri::AppHandle,
    ) -> Result<RunId, String> {
        let mut cmd = Command::new(binary);
        cmd.args(args);
        cmd.stdin(std::process::Stdio::piped());
        cmd.stdout(std::process::Stdio::piped());
        cmd.stderr(std::process::Stdio::piped());

        unsafe {
            cmd.pre_exec(|| {
                libc::setsid();
                Ok(())
            });
        }

        let mut child = cmd.spawn().map_err(|e| e.to_string())?;
        let pid = child.id().ok_or("no child pid")?;
        let pgid = pid as i32;

        let run_id = self.next_id.fetch_add(1, Ordering::Relaxed);

        // Write prompt to stdin
        if let Some(stdin) = child.stdin.as_mut() {
            use tokio::io::AsyncWriteExt;
            stdin.write_all(prompt.as_bytes()).await.map_err(|e| e.to_string())?;
            stdin.flush().await.map_err(|e| e.to_string())?;
        }
        // Drop stdin to signal EOF to the CLI
        child.stdin.take();

        let run_id_clone = run_id;
        let app_handle_clone = app_handle.clone();
        let stdout = child.stdout.take().unwrap();

        self.runs.lock().unwrap().insert(run_id, RunState { child, pgid });

        tokio::spawn(async move {
            let reader = tokio::io::BufReader::new(stdout);
            let mut lines = reader.lines();
            while let Ok(Some(line)) = lines.next_line().await {
                let _ = app_handle_clone.emit("ai:chunk", AiChunk {
                    run_id: run_id_clone,
                    text: line,
                });
            }
            let _ = app_handle_clone.emit("ai:done", AiDone { run_id: run_id_clone });
        });

        Ok(run_id)
    }

    pub async fn run_blocking(
        &self,
        binary: &str,
        args: &[String],
        prompt: &str,
    ) -> Result<String, String> {
        let mut cmd = Command::new(binary);
        cmd.args(args);
        cmd.stdin(std::process::Stdio::piped());
        cmd.stdout(std::process::Stdio::piped());
        cmd.stderr(std::process::Stdio::piped());

        let mut child = cmd.spawn().map_err(|e| e.to_string())?;

        if let Some(stdin) = child.stdin.as_mut() {
            use tokio::io::AsyncWriteExt;
            stdin.write_all(prompt.as_bytes()).await.map_err(|e| e.to_string())?;
            stdin.flush().await.map_err(|e| e.to_string())?;
        }
        child.stdin.take();

        let output = child.wait_with_output().await.map_err(|e| e.to_string())?;
        let stdout = String::from_utf8_lossy(&output.stdout).to_string();
        Ok(stdout)
    }

    pub fn cancel(&self, run_id: RunId) -> Result<(), String> {
        let state = self.runs.lock().unwrap().remove(&run_id).ok_or("run not found")?;
        let result = unsafe { libc::killpg(state.pgid, libc::SIGTERM) };
        if result != 0 {
            return Err("failed to kill process group".into());
        }
        let pgid = state.pgid;
        tokio::spawn(async move {
            tokio::time::sleep(std::time::Duration::from_secs(2)).await;
            unsafe { libc::killpg(pgid, libc::SIGKILL); }
        });
        Ok(())
    }
}
```

```rust
mod adapter;
mod detector;
mod runner;

use std::sync::Arc;
use tauri::plugin::{Builder as PluginBuilder, TauriPlugin};
use tauri::{AppHandle, Emitter, Manager, Runtime};
use zeep_core::prefs::PrefsState;

use adapter::{get_adapter, RunOpts};
use detector::DetectedProvider;
use runner::AIRunner;

const PREFS_KEY_PROVIDERS: &str = "zeep_ai.detected_providers";

pub struct Builder;

impl Builder {
    pub fn new() -> Self { Self }

    pub fn build<R: Runtime>() -> TauriPlugin<R> {
        PluginBuilder::<R>::new("zeep-ai")
            .setup(|app, _api| {
                let runner = Arc::new(AIRunner::new());
                app.manage(runner);
                Ok(())
            })
            .invoke_handler(tauri::generate_handler![
                ai_detect_providers,
                ai_run,
                ai_run_stream,
                ai_cancel,
            ])
            .build()
    }
}

#[tauri::command]
async fn ai_detect_providers(prefs: tauri::State<'_, PrefsState>) -> Result<Vec<DetectedProvider>, String> {
    let providers = detector::detect();

    // Cache in PrefsStore
    if let Ok(json) = serde_json::to_value(&providers) {
        if let Ok(prefs) = prefs.write() {
            prefs.set(PREFS_KEY_PROVIDERS.to_string(), json);
        }
    }

    Ok(providers)
}

#[tauri::command]
async fn ai_run(
    app: AppHandle,
    provider: String,
    prompt: String,
    model: Option<String>,
    system: Option<String>,
) -> Result<String, String> {
    let adapter = get_adapter(&provider).ok_or_else(|| format!("unknown provider: {provider}"))?;
    let opts = RunOpts { model, max_tokens: None, system };

    // Get binary from prefs cache
    let prefs = app.state::<PrefsState>();
    let binary = get_binary_from_prefs(&prefs, &provider)?;

    let args = adapter.build_args(&opts);
    let runner = app.state::<Arc<AIRunner>>();
    runner.run_blocking(&binary, &args, &prompt).await
}

#[tauri::command]
async fn ai_run_stream(
    app: AppHandle,
    provider: String,
    prompt: String,
    model: Option<String>,
    system: Option<String>,
) -> Result<u64, String> {
    let adapter = get_adapter(&provider).ok_or_else(|| format!("unknown provider: {provider}"))?;
    let opts = RunOpts { model, max_tokens: None, system };

    let prefs = app.state::<PrefsState>();
    let binary = get_binary_from_prefs(&prefs, &provider)?;

    let args = adapter.build_args(&opts);
    let runner = app.state::<Arc<AIRunner>>();
    runner.run_stream(&binary, &args, &prompt, app).await
}

#[tauri::command]
async fn ai_cancel(app: AppHandle, run_id: u64) -> Result<(), String> {
    let runner = app.state::<Arc<AIRunner>>();
    runner.cancel(run_id)
}

fn get_binary_from_prefs(prefs: &PrefsState, provider: &str) -> Result<String, String> {
    let prefs = prefs.read().map_err(|e| e.to_string())?;
    let providers: Vec<DetectedProvider> = prefs.get(PREFS_KEY_PROVIDERS)
        .and_then(|v| serde_json::from_value(v).ok())
        .ok_or_else(|| format!("providers not detected yet. Call ai_detect_providers first"))?;
    providers.into_iter()
        .find(|p| p.name == provider)
        .map(|p| p.binary)
        .ok_or_else(|| format!("provider {provider} not found"))
}
```

```typescript
export function stripFences(text: string): string {
  return text.replace(/^```[\s\S]*?\n/, '').replace(/\n```$/, '').trim();
}

export function firstLine(text: string): string {
  return text.split('\n').find(l => l.trim().length > 0) ?? '';
}

export function truncate(text: string, maxChars: number): string {
  if (text.length <= maxChars) return text;
  const trimmed = text.slice(0, maxChars);
  const lastSpace = trimmed.lastIndexOf(' ');
  return (lastSpace > 0 ? trimmed.slice(0, lastSpace) : trimmed) + '…';
}

export function extractJson(text: string): unknown {
  const match = text.match(/```(?:json)?\s*([\s\S]*?)```/) ?? text.match(/{[\s\S]*?}/);
  if (!match) throw new Error('no JSON found');
  return JSON.parse(match[1] ?? match[0]);
}
```
